```kotlin
package com.il2cpp.scanner.data.model

import androidx.room.Entity
import androidx.room.PrimaryKey
import androidx.room.TypeConverter
import androidx.room.TypeConverters
import com.google.gson.Gson
import com.google.gson.reflect.TypeToken
import java.util.UUID

enum class TargetKind {
    FIELD,   // Field Offset (e.g. 0x18)
    METHOD   // Method RVA (e.g. 0x1A42F0)
}

enum class CodeStylePreset(val label: String, val template: String) {
    CPP_CONSTEXPR("C++ (constexpr)", "constexpr uintptr_t {name} = {offset};"),
    CPP_MACRO("C++ (#define)", "#define {name} {offset}"),
    CS_STATIC("C# (const uint)", "public const uint {name} = {offset};"),
    STRUCT_HEADER("Struct / Header", "    uintptr_t {name}; // Offset: {offset}"),
    RAW_JSON("JSON Map", "  \"{name}\": \"{offset}\","),
    CSV("CSV Format", "{class},{name},{offset},{kind}"),
    CUSTOM("Custom Template", "constexpr uintptr_t {name} = {offset};")
}

data class TargetItem(
    val id: String = UUID.randomUUID().toString(),
    val customName: String = "",
    val className: String,
    val memberName: String,
    val kind: TargetKind = TargetKind.FIELD,
    val comment: String = "",
    val fallbackClasses: List<String> = emptyList(),
    val fallbackMembers: List<String> = emptyList(),
    // Resolved Scan Metadata
    val resolvedOffset: String? = null,
    val resolvedSignature: String? = null,
    val resolvedViaFallback: Boolean = false,
    val resolvedClassName: String? = null,
    val resolvedMemberName: String? = null,
    val isFound: Boolean = false
)

@Entity(tableName = "watchlist_profiles")
data class WatchlistProfile(
    @PrimaryKey val id: String = UUID.randomUUID().toString(),
    val name: String,
    val description: String = "",
    val codeStyle: CodeStylePreset = CodeStylePreset.CPP_CONSTEXPR,
    val customTemplate: String = "constexpr uintptr_t {name} = {offset};",
    val items: List<TargetItem> = emptyList(),
    val isDefault: Boolean = false,
    val createdAt: Long = System.currentTimeMillis()
)

@Entity(tableName = "scan_history")
data class ScanHistoryRecord(
    @PrimaryKey val id: String = UUID.randomUUID().toString(),
    val timestamp: Long = System.currentTimeMillis(),
    val dumpFileName: String,
    val profileName: String,
    val profileId: String,
    val totalTargets: Int,
    val matchedCount: Int,
    val fallbackCount: Int,
    val missingCount: Int,
    val generatedCode: String,
    val items: List<TargetItem> = emptyList()
)

// Room Type Converters
class RoomConverters {
    private val gson = Gson()

    @TypeConverter
    fun fromTargetItemList(value: List<TargetItem>): String = gson.toJson(value)

    @TypeConverter
    fun toTargetItemList(value: String): List<TargetItem> {
        val type = object : TypeToken<List<TargetItem>>() {}.type
        return gson.fromJson(value, type) ?: emptyList()
    }
}
```

```kotlin
package com.il2cpp.scanner.util

import com.il2cpp.scanner.data.model.CodeStylePreset
import com.il2cpp.scanner.data.model.TargetItem
import com.il2cpp.scanner.data.model.TargetKind

object CodeFormatter {

    fun formatSnippet(
        item: TargetItem,
        preset: CodeStylePreset,
        customTemplate: String = "constexpr uintptr_t {name} = {offset};"
    ): String {
        val template = when (preset) {
            CodeStylePreset.CPP_CONSTEXPR -> "constexpr uintptr_t {name} = {offset};"
            CodeStylePreset.CPP_MACRO      -> "#define {name} {offset}"
            CodeStylePreset.CS_STATIC      -> "public const uint {name} = {offset};"
            CodeStylePreset.STRUCT_HEADER  -> "    uintptr_t {name}; // Offset: {offset}"
            CodeStylePreset.RAW_JSON       -> "  \"{name}\": \"{offset}\","
            CodeStylePreset.CSV            -> "{class},{name},{offset},{kind}"
            CodeStylePreset.CUSTOM         -> customTemplate
        }

        val displayName = item.customName.ifBlank { item.memberName }
        val safeName = displayName.replace(Regex("[^a-zA-Z0-9_]"), "_")
        val offset = item.resolvedOffset ?: "0x0"
        val targetClass = item.resolvedClassName ?: item.className
        val kindStr = if (item.kind == TargetKind.FIELD) "Field" else "Method"

        return template
            .replace("{name}", safeName)
            .replace("{rawName}", displayName)
            .replace("{memberName}", item.memberName)
            .replace("{class}", targetClass)
            .replace("{offset}", offset)
            .replace("{rva}", offset)
            .replace("{kind}", kindStr)
            .replace("{comment}", item.comment)
    }

    fun generateFullExport(
        profileName: String,
        preset: CodeStylePreset,
        items: List<TargetItem>,
        customTemplate: String = ""
    ): String {
        val safeProfileName = profileName.replace(Regex("[^a-zA-Z0-9_]"), "")
        val builder = StringBuilder()

        when (preset) {
            CodeStylePreset.CPP_CONSTEXPR -> {
                builder.appendLine("// =========================================================")
                builder.appendLine("// Generated by IL2CPP Scanner - Profile: $profileName")
                builder.appendLine("// Total targets: ${items.size}")
                builder.appendLine("// =========================================================")
                builder.appendLine("#pragma once")
                builder.appendLine("#include <cstdint>\n")
                builder.appendLine("namespace Offsets {")
                builder.appendLine("    namespace $safeProfileName {")
                items.forEach { item ->
                    if (item.comment.isNotBlank()) {
                        builder.appendLine("        // ${item.comment}")
                    }
                    builder.appendLine("        " + formatSnippet(item, preset, customTemplate))
                }
                builder.appendLine("    }")
                builder.appendLine("}")
            }
            CodeStylePreset.CS_STATIC -> {
                builder.appendLine("// Generated by IL2CPP Scanner - Profile: $profileName")
                builder.appendLine("public static class ${safeProfileName}Offsets {")
                items.forEach { item ->
                    if (item.comment.isNotBlank()) {
                        builder.appendLine("    // ${item.comment}")
                    }
                    builder.appendLine("    " + formatSnippet(item, preset, customTemplate))
                }
                builder.appendLine("}")
            }
            CodeStylePreset.STRUCT_HEADER -> {
                builder.appendLine("// Struct layout for: $safeProfileName")
                builder.appendLine("struct ${safeProfileName}Fields {")
                items.forEach { item ->
                    builder.appendLine("    " + formatSnippet(item, preset, customTemplate))
                }
                builder.appendLine("};")
            }
            CodeStylePreset.RAW_JSON -> {
                builder.appendLine("{")
                items.forEachIndexed { index, item ->
                    val comma = if (index < items.size - 1) "" else ""
                    builder.appendLine("  " + formatSnippet(item, preset, customTemplate))
                }
                builder.appendLine("}")
            }
            else -> {
                items.forEach { item ->
                    builder.appendLine(formatSnippet(item, preset, customTemplate))
                }
            }
        }
        return builder.toString()
    }
}
```

6. DUMP PARSER & MATCHING ENGINE (DumpParserEngine.kt)
==============================================================================

```kotlin
package com.il2cpp.scanner.engine

import com.il2cpp.scanner.data.model.TargetItem
import com.il2cpp.scanner.data.model.TargetKind
import kotlinx.coroutines.Dispatchers
import kotlinx.coroutines.withContext
import java.io.BufferedReader
import java.io.InputStream
import java.io.InputStreamReader

class DumpParserEngine {

    suspend fun scanDump(
        inputStream: InputStream,
        targets: List<TargetItem>,
        onProgress: (Float) -> Unit = {}
    ): List<TargetItem> = withContext(Dispatchers.IO) {
        val lines = mutableListOf<String>()
        BufferedReader(InputStreamReader(inputStream)).use { reader ->
            var line: String?
            while (reader.readLine().also { line = it } != null) {
                lines.add(line!!)
            }
        }

        val offsetRegex = Regex("""// Offset:\s*(0x[0-9a-fA-F]+)""")
        val rvaRegex    = Regex("""// RVA:\s*(0x[0-9a-fA-F]+)""")

        val resolvedTargets = targets.mapIndexed { index, target ->
            onProgress(index.toFloat() / targets.size)

            var matchFound: Pair<String, String>? = null
            var resolvedViaFallback = false
            var resolvedClass = target.className
            var resolvedMember = target.memberName

            // 1. Primary Search
            matchFound = findInLines(lines, target.className, target.memberName, target.kind, offsetRegex, rvaRegex)

            // 2. Fallback Search if primary not found
            if (matchFound == null) {
                val candidateClasses = listOf(target.className) + target.fallbackClasses
                val candidateMembers = listOf(target.memberName) + target.fallbackMembers

                searchLoop@ for (c in candidateClasses) {
                    for (m in candidateMembers) {
                        val fbMatch = findInLines(lines, c, m, target.kind, offsetRegex, rvaRegex)
                        if (fbMatch != null) {
                            matchFound = fbMatch
                            resolvedViaFallback = true
                            resolvedClass = c
                            resolvedMember = m
                            break@searchLoop
                        }
                    }
                }
            }

            if (matchFound != null) {
                target.copy(
                    isFound = true,
                    resolvedOffset = matchFound.first,
                    resolvedSignature = matchFound.second,
                    resolvedViaFallback = resolvedViaFallback,
                    resolvedClassName = resolvedClass,
                    resolvedMemberName = resolvedMember
                )
            } else {
                target.copy(
                    isFound = false,
                    resolvedOffset = null,
                    resolvedSignature = null,
                    resolvedViaFallback = false
                )
            }
        }

        onProgress(1.0f)
        resolvedTargets
    }

    private fun findInLines(
        lines: List<String>,
        className: String,
        memberName: String,
        kind: TargetKind,
        offsetRegex: Regex,
        rvaRegex: Regex
    ): Pair<String, String>? {
        var insideClass = false

        for (i in lines.indices) {
            val line = lines[i]

            if (line.contains("class $className") || line.contains("struct $className")) {
                insideClass = true
            }

            if (insideClass && line.contains(memberName)) {
                val prevLine = if (i > 0) lines[i - 1] else ""
                if (kind == TargetKind.FIELD) {
                    val match = offsetRegex.find(prevLine) ?: offsetRegex.find(line)
                    if (match != null) {
                        return Pair(match.groupValues[1], line.trim())
                    }
                } else {
                    val match = rvaRegex.find(prevLine) ?: rvaRegex.find(line)
                    if (match != null) {
                        return Pair(match.groupValues[1], line.trim())
                    }
                }
            }

            if (insideClass && line.startsWith("}")) {
                // Class boundary exit heuristic
            }
        }
        return null
    }
}
```

```kotlin
                            icon = { Icon(Icons.Rounded.Close, contentDescription = null, modifier = Modifier.size(12.dp)) }
                        )
                    }
                }

                // Fallback Member Names
                Text(text = "Fallback Field / Method Names", fontSize = 11.sp, fontWeight = FontWeight.Medium, color = TextLight)
                Row(modifier = Modifier.fillMaxWidth(), horizontalArrangement = Arrangement.spacedBy(6.dp)) {
                    OutlinedTextField(
                        value = memberInput,
                        onValueChange = { memberInput = it },
                        placeholder = { Text("e.g. m_Speed", fontSize = 11.sp, color = TextSecondary) },
                        modifier = Modifier.weight(1f),
                        singleLine = true,
                        keyboardOptions = KeyboardOptions(imeAction = ImeAction.Done),
                        keyboardActions = KeyboardActions(onDone = {
                            if (memberInput.isNotBlank()) {
                                onAddMember(memberInput.trim())
                                memberInput = ""
                            }
                        })
                    )
                    Button(
                        onClick = {
                            if (memberInput.isNotBlank()) {
                                onAddMember(memberInput.trim())
                                memberInput = ""
                            }
                        },
                        colors = ButtonDefaults.buttonColors(containerColor = PrimaryIndigo)
                    ) {
                        Text("+", fontSize = 14.sp)
                    }
                }

                // Member Chips
                Row(modifier = Modifier.fillMaxWidth(), horizontalArrangement = Arrangement.spacedBy(6.dp)) {
                    fallbackMembers.forEach { tag ->
                        SuggestionChip(
                            onClick = { onRemoveMember(tag) },
                            label = { Text(tag, fontFamily = FontFamily.Monospace, fontSize = 11.sp) },
                            icon = { Icon(Icons.Rounded.Close, contentDescription = null, modifier = Modifier.size(12.dp)) }
                        )
                    }
                }
            }
        }
    }
}
```

App Header (Top Bar):
- Positioned fixed at the very top.

Main Navigation (Bottom Bar):
- Positioned fixed at the very bottom of the screen.
- 3 evenly spaced tabs: "Target Setup", "Profile", and "History".

Tab 1: Target Setup (Main Scanning Screen):
- Vertically stacked cards with standard padding.
- Top Box: Dump file upload area with a dashed border, centered icon, and text. Changes to a solid state with file size and a trash icon on the right when loaded.
- Middle Box: Active profile selector. Features a 2dp blue gradient line on its left edge. Contains a dropdown, target count badge, and a prominent, full-width gradient "Run Offset Scan" button at the bottom of the card.
- Bottom Section (Post-Scan): 
  - A horizontal scrolling row of 3 stat chips (Matched in green, Fallback in amber, Missing in red).
  - Code preview block with a dropdown at the top right to change languages, and Copy/Save buttons.
  - A vertical list of scanned target cards. Each card has a kind badge (Field/Method) on the left, target names in the center, and the resolved hex offset on the right.

Tab 2: Profiles & Target Lists:
- Overview Screen:
  - Top section has the title on the left, and "+ New Profile", "Import", "Export" buttons aligned to the right.
  - Profile Cards stack vertically. Each card shows the name, description, and style badge. The bottom row of the card holds 4 quick action buttons (Select, Edit, Duplicate, Delete).

  - Top bar changes to show a "< Back to Profiles" button on the left, the profile name, and a "+ Add Target" button on the right.

Tab 3: Scan History:
- Top header shows the scan count and a red "Clear All History" button on the right.

Modals & Dialogs (Add Target, Edit Profile, Code View):
- Background: Dimmed translucent black overlay spanning the entire screen.
- Desktop Positioning: Centered perfectly in the middle of the screen.
- Mobile Positioning: Anchored slightly towards the top to prevent the on-screen keyboard from covering the "Save" buttons.
- Internal Layout: Inputs and labels are stacked vertically with consistent gap spacing.
- Fallbacks Section (in Target Modals): A collapsible row. When clicked, it expands downwards to reveal text inputs and horizontally wrapping "chips" for the fallback tags.
